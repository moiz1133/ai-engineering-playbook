## Final Fixed Code
```python
def find_users_by_department(users: list, department: str) -> list:
    if not isinstance(department, str) or not department:
        raise ValueError("Invalid department string")
    
    # Create an index for users by department
    department_index = {}
    for user in users:
        if 'department' in user:
            dept = user['department']
            if dept not in department_index:
                department_index[dept] = []
            department_index[dept].append(user)
    
    return department_index.get(department, [])

def get_user_email(users: list, user_id: int) -> str:
    user_dict = {user['id']: user for user in users if 'id' in user}
    
    user = user_dict.get(user_id)
    if user is not None:
        return user['email']
    else:
        raise ValueError("User not found")
```
